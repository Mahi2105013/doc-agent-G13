import logging

logger = logging.getLogger(__name__)

try:
    import cv2
except ImportError:
    cv2 = None

def preprocess_image(image: np.ndarray) -> np.ndarray:
    """Enhance image for OCR using OpenCV if available.
    
    If cv2 is not installed, gracefully returns the original image.
    Applies grayscale conversion and Otsu's thresholding.
    """
    if cv2 is None:
        logger.warning("cv2 not installed. Skipping preprocessing.")
        return image
        
    try:
        # Convert to grayscale
        if len(image.shape) == 3:
            gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        else:
            gray = image

        # It assumes a bimodal distribution of pixel intensities
        _, thresh = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    """
    Detect text regions in a document image using YOLO.
    Ignores headers, footers, and images.
    Returns a list of bounding boxes (x1, y1, x2, y2).

import pytesseract

from pathlib import Path
from pydantic import BaseModel
import logging

logger = logging.getLogger(__name__)

class TesseractConfig(BaseModel):
    tessdata_dir: str = "src/doc_agent/models"
    lang: str = "ben_seerah"
    psm: int = 6  # Assume a single uniform block of text

class TesseractReader:
    def __init__(self, config: TesseractConfig):
        self.config = config
        self.tessdata_path = Path(config.tessdata_dir).resolve()
        
        if not (self.tessdata_path / f"{config.lang}.traineddata").exists():
            logger.warning(f"Traineddata for {config.lang} not found at {self.tessdata_path}")
            
    def transcribe(self, image: Image.Image, regions: list[tuple[int, int, int, int]]) -> str:
        """
        Transcribe specific regions of an image using Tesseract.
        """
        if not regions:
            # If no regions detected, OCR the whole image
            regions = [(0, 0, image.width, image.height)]
            
        # Point Tesseract to our custom model directory
        custom_oem_psm_config = f'--tessdata-dir "{self.tessdata_path}" --psm {self.config.psm}'
        
        extracted_texts = []
        for box in regions:
            try:
                # Crop the region identified by layout detection
                cropped_img = image.crop(box)
                text = pytesseract.image_to_string(
                    cropped_img, 
                    lang=self.config.lang, 
                    config=custom_oem_psm_config
                )
                if text.strip():
                    extracted_texts.append(text.strip())
            except Exception as e:
                logger.error(f"OCR failed for region {box}: {e}")
                
        # Join paragraphs with double newlines
        return "\n\n".join(extracted_texts)

Me: Now that we have the extracted text, we need to chunk it in `src/doc_agent/index/chunk.py` before embedding. What's the best strategy for a Bengali book?

AI: You should use **Recursive Character Splitting with Overlap**.

*Analogy time:* Think of chunking like cutting a pizza. If you cut the slices too big (entire pages), they won't fit on the plate (the LLM's context window). If you cut them into tiny 1-inch squares (individual words), you ruin the toppings and lose the flavor (semantic context). Recursive chunking cuts normal slices but leaves a little extra crust overlapping the next slice, ensuring you never cut a pepperoni (a critical sentence or thought) in half.

For Bengali, you must respect the structure: paragraphs first, then the Bengali sentence ending punctuation (দাড়ি '।'), then words.

AI: Absolutely. Keeping strict data contracts is vital. Here is a sophisticated recursive chunker for `chunk.py` that handles the Bengali `।` character and adheres to your Pydantic schema:

```python
from pydantic import BaseModel, Field

class ChunkConfig(BaseModel):
    chunk_size: int = 500
    overlap: int = 50

class Chunk(BaseModel):
    text: str
    doc_id: str
    page_ids: list[str] = Field(default_factory=list)

def recursive_chunk_text(text: str, config: ChunkConfig, doc_id: str, page_id: str) -> list[Chunk]:
    """
    Recursively split text trying \n\n, then \n, then Bengali '।', then spaces.
    Outputs strict Chunk objects.
    """
    if not text.strip():
        return []
        
    if len(text) <= config.chunk_size:
        return [Chunk(text=text, doc_id=doc_id, page_ids=[page_id])]
        
    separators = ["\n\n", "\n", "।", " "]
    
    for sep in separators:
        splits = text.split(sep)
        if len(splits) > 1:
            chunks = []
            current_chunk = ""
            
            for split in splits:
                # Calculate proposed length, including separator if needed
                proposed_len = len(current_chunk) + len(sep) + len(split) if current_chunk else len(split)
                
                if proposed_len <= config.chunk_size:
                    current_chunk += (sep + split) if current_chunk else split
                else:
                    if current_chunk:
                        chunks.append(current_chunk)
                    # Start new chunk with overlap logic (simplified here)
                    current_chunk = split
                    
            if current_chunk:
                chunks.append(current_chunk)
                
            # If we successfully split, return the Chunks
            if len(chunks) > 1 or len(chunks[0]) < len(text):
                return [Chunk(text=c.strip(), doc_id=doc_id, page_ids=[page_id]) 
                        for c in chunks if c.strip()]
            
    # Fallback: Hard slice if no separators work (rare, but prevents infinite loops)
    return [
        Chunk(text=text[i:i + config.chunk_size], doc_id=doc_id, page_ids=[page_id]) 
        for i in range(0, len(text), config.chunk_size - config.overlap)
    ]
```

import logging
from pydantic import BaseModel

logger = logging.getLogger(__name__)

class EmbedConfig(BaseModel):
    model_name: str = "sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2"
    batch_size: int = 32
    normalize: bool = True

class Embedder:
    def __init__(self, config: EmbedConfig):
        self.config = config
        self.model = None
        self._load_model()
        
    def _load_model(self):
        try:
            from sentence_transformers import SentenceTransformer
            logger.info(f"Loading embedding model: {self.config.model_name}")
            self.model = SentenceTransformer(self.config.model_name)
        except ImportError:
            logger.error("sentence-transformers not installed. Using dummy embeddings.")
            self.model = None

    def encode(self, texts: list[str]) -> np.ndarray:
        """Embed a list of text chunks."""
        if not texts:
            return np.empty((0, 384), dtype=np.float32)
            
        if self.model is None:
            # Dummy fallback for CI/CD structural tests
            logger.warning("Generating random embeddings! Do not use in production.")
            embeddings = np.random.randn(len(texts), 384).astype(np.float32)
            if self.config.normalize:
                norms = np.linalg.norm(embeddings, axis=1, keepdims=True)
                embeddings = embeddings / norms
            return embeddings

        embeddings = self.model.encode(
            texts, 
            batch_size=self.config.batch_size,
            show_progress_bar=False, 
            normalize_embeddings=self.config.normalize
        )
        return np.asarray(embeddings, dtype=np.float32)

Me: Now we need to store these 384-dimensional vectors in `src/doc_agent/index/store.py`. I've heard of FAISS. Should I use Flat L2 or HNSW?

AI: FAISS (Facebook AI Similarity Search) is perfect here. It's the industry standard for fast, dense vector retrieval.

Because you applied `normalize=True` in the embedding step, your vectors lie on a unit hypersphere. Therefore, you should use **Inner Product (IP)**, which mathematically equivalent to Cosine Similarity for normalized vectors but computes much faster.

* **FlatIP (`faiss.IndexFlatIP`):** Compares the query against *every single chunk*. 100% accurate (exact nearest neighbor), but slow for millions of vectors.
* **HNSW (`faiss.IndexHNSWFlat`):** A graph-based index. It's incredibly fast (approximate nearest neighbors).

Since this is a single book (a few thousand chunks), FlatIP is perfectly fine and guaranteed to return the true top-K. However, HNSW is industry-standard if you plan to scale up to the entire library. Let's write the code to support both, defaulting to HNSW.

Me: Great, write the `store.py` logic. It needs to keep track of the text payloads too, not just the vectors, so we can actually return the text during a search!

AI: Spot on. FAISS only stores the vectors (and their integer IDs). We need an auxiliary data structure to map those IDs back to the actual `Chunk` payloads. Here is a complete FAISS index manager:

```python
import faiss
import numpy as np
from pydantic import BaseModel
from typing import Optional

# Assume Chunk is imported from chunk.py
# from .chunk import Chunk

class StoreConfig(BaseModel):
    index_type: str = "hnsw" # or "flat"
    dim: int = 384
    metric: str = "cosine" # Uses Inner Product under the hood for normalized vectors

class VectorStore:
    def __init__(self, config: StoreConfig):
        self.config = config
        self.chunks: list = [] # To store the actual payloads
        
        if config.index_type.lower() == "hnsw":
            # HNSW32 means each node is connected to 32 neighbors
            self.index = faiss.IndexHNSWFlat(config.dim, 32, faiss.METRIC_INNER_PRODUCT)
        else:
            self.index = faiss.IndexFlatIP(config.dim)
            
    def add(self, chunks: list, embeddings: np.ndarray):
        """Add vectors and their corresponding text chunks to the store."""
        if len(chunks) != embeddings.shape[0]:
            raise ValueError("Number of chunks must match number of embeddings.")
            
        self.index.add(embeddings)
        self.chunks.extend(chunks)
        
    def search(self, query_embedding: np.ndarray, top_k: int = 3) -> list[tuple[float, object]]:
        """Retrieve the top K chunks for a query vector."""
        if self.index.ntotal == 0:
            return []
            
        # Ensure query is 2D: (1, dim)
        if len(query_embedding.shape) == 1:
            query_embedding = np.expand_dims(query_embedding, axis=0)
            
        distances, indices = self.index.search(query_embedding, top_k)
        
        results = []
        for dist, idx in zip(distances[0], indices[0]):
            if idx != -1 and idx < len(self.chunks):
                results.append((float(dist), self.chunks[idx]))
                
        return results
```
